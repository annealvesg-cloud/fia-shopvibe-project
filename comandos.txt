antifraude_db:

    docker exec -it postgres_antifraude psql -U postgres -d antifraude_db

    SELECT pedido_id, cliente_id, valor, motivo, score, detectado_em
    FROM transacoes_suspeitas
    ORDER BY id DESC
    LIMIT 10;



analytics_db (camada gold):

    docker exec -it postgres_analytics psql -U postgres -d analytics_db

    SELECT
    pedido_id,
    cliente_id,
    valor,
    status,
    criado_em,
    source_ts
    FROM gold_pedidos
    ORDER BY criado_em;

    SELECT *
    FROM gold_indicadores_diarios
    ORDER BY data;

    SELECT *
    FROM gold_processamento
    ORDER BY processado_em;
